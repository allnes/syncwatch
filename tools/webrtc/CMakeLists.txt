cmake_minimum_required(VERSION 3.15)
project(syncwatch_remote_media_check LANGUAGES CXX)
set(WEBRTC_INCLUDE_DIR "" CACHE PATH "libwebrtc include directory")
option(ENABLE_TSAN "Enable ThreadSanitizer with Clang" OFF)
find_package(Threads REQUIRED)
add_executable(check_remote_media_cache check_remote_media_cache.cpp)
target_compile_features(check_remote_media_cache PRIVATE cxx_std_17)
target_include_directories(check_remote_media_cache PRIVATE "${WEBRTC_INCLUDE_DIR}")
target_link_libraries(check_remote_media_cache PRIVATE Threads::Threads)
if(ENABLE_TSAN)
  target_compile_options(check_remote_media_cache PRIVATE -fsanitize=thread -g)
  target_link_options(check_remote_media_cache PRIVATE -fsanitize=thread)
endif()
enable_testing()
add_test(NAME remote_media_cache COMMAND check_remote_media_cache)
set_tests_properties(remote_media_cache PROPERTIES TIMEOUT 60)
